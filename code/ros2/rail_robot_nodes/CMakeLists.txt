cmake_minimum_required(VERSION 3.20)
project(rail_robot_nodes)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
# Keep one source copy shared with the existing Python launcher.
set(JETSON_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../jetson_code")
ament_python_install_package(node_runtime PACKAGE_DIR "${JETSON_SOURCE}/node_runtime")
# Existing TensorRT/CUDA and report libraries use top-level imports.
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
file(GLOB BACKEND_MODULES "${JETSON_SOURCE}/*.py")
list(FILTER BACKEND_MODULES EXCLUDE REGEX "/(run|training_capture|dc_test)\\.py$")
install(FILES ${BACKEND_MODULES} DESTINATION "${PYTHON_INSTALL_DIR}")
foreach(ROLE camera vision control stm_bridge mission inspection display upload training_capture capture_debug actuator_debug sensor_debug firebase_snapshot gui_probe webrtc_sender monitor record dashboard_telemetry remote_control audio_alert frame_profiler)
  configure_file(node_executable.in "${CMAKE_CURRENT_BINARY_DIR}/${ROLE}_node" @ONLY
    FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/${ROLE}_node" DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
